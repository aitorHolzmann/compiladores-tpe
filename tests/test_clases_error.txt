class c_base
begin
    shortint atr_privado;
    shortint atr_exportado export to c_autorizada;

    shortint m_privado()
    begin
        ret(1$s);
    end;

    shortint m_exportado()
    begin
        ret(2$s);
    end export to c_autorizada;
end;

// Atributo y metodo redeclarados en la misma clase
class c_redeclaraciones
begin
    shortint atr_dup;
    shortint atr_dup;

    shortint m_dup()
    begin
        ret(1$s);
    end;

    shortint m_dup()
    begin
        ret(2$s);
    end;
end;

// Clase base no declarada en EXTENDS
class c_herencia_invalida
begin
    extends c_fantasma;
    shortint x;
end;

// Clase no declarada en IMPORT FROM
class c_import_invalido import from c_inexistente
begin
    shortint y;
end;

class c_autorizada import from c_base
begin
    shortint m_test(c_base obj)
    begin
        // Miembro no exportado a esta clase
        obj[0$s] := 10$s;
        obj.m_privado();

        ret(0$s);
    end;
end;

class c_no_importa
begin
    shortint m_test(c_base obj)
    begin
        // La clase c_no_importa no importa a c_base
        obj[1$s] := 20$s;
        obj.m_exportado();

        ret(0$s);
    end;
end;

// Herencia directa a si misma
class c_auto_ciclo
begin
    extends c_auto_ciclo;
    shortint x;
end;

// Intento de ciclo mutuo por referencia hacia adelante
class c_fwd_a
begin
    extends c_fwd_b;
    shortint atr_a;
end;

class c_fwd_b
begin
    extends c_fwd_a;
    shortint atr_b;
end;

// Ciclo indirecto mediante redeclaracion de clase
class c_ciclo1
begin
    shortint val1;
end;

class c_ciclo2
begin
    extends c_ciclo1;
    shortint val2;
end;

class c_ciclo1
begin
    extends c_ciclo2;
    shortint val3;
end;

c_base b;
shortint r_entero;
singlef idx_float;
c_fwd_a obj_fwd;

// Objeto de clase no declarada
c_no_existe obj_malo;

begin
    // Indice posicional fuera de rango
    b[5$s] := 1$s;

    // Indice de acceso posicional no es SHORTINT
    idx_float := 1.5;
    b[idx_float] := 1$s;

    // Metodo no encontrado en la jerarquia
    b.metodo_inexistente();

    // Acceso e invocacion sobre identificador que no es objeto
    r_entero := 5$s;
    r_entero[0$s] := 1$s;
    r_entero.m();

    // Incompatibilidad de tipos en asignacion a atributo
    b[0$s] := 3.14;

    // Metodo inexistente sobre clase con intento de ciclo mutuo
    obj_fwd.metodo_inexistente();
end
