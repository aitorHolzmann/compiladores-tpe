programa_clases_ok

class c_base1
begin
    shortint b1_atr0 export to c_cliente;
    singlef b1_atr1 export to c_cliente;

    shortint m_base1()
    begin
        ret(b1_atr0);
    end export to c_cliente;
end;

class c_base2
begin
    shortint b2_atr0;

    shortint m_base2(shortint x)
    begin
        ret(x + 1$s);
    end;
end;

class c_derivada
begin
    extends c_base1, c_base2;
    shortint der_atr0;

    shortint m_propio()
    begin
        ret(100$s);
    end;
end;

class c_cliente import from c_base1
begin
    shortint m_cliente(c_base1 obj_b1)
    begin
        obj_b1[0$s] := 10$s;
        obj_b1.m_base1();
        ret(0$s);
    end;
end;

// Jerarquia en diamante para probar herencia multiple y desambiguacion por orden
class c_raiz
begin
    shortint r_val;

    shortint m_raiz()
    begin
        ret(10$s);
    end;

    shortint m_comun()
    begin
        ret(1$s);
    end;
end;

class c_rama1
begin
    extends c_raiz;
    shortint r1_val;

    shortint m_comun()
    begin
        ret(20$s);
    end;
end;

class c_rama2
begin
    extends c_raiz;
    shortint r2_val;

    shortint m_comun()
    begin
        ret(30$s);
    end;

    shortint m_rama2()
    begin
        ret(40$s);
    end;
end;

class c_diamante
begin
    extends c_rama1, c_rama2;
    shortint d_val;

    shortint m_propio_dia()
    begin
        ret(50$s);
    end;
end;

c_derivada d;
c_base1 b;
c_cliente cli;
c_diamante dia;
shortint r_short, idx;
singlef r_single;

begin
    // Acceso posicional con constante
    b[0$s] := 25$s;
    b[1$s] := 3.14;
    r_short := b[0$s];
    r_single := b[1$s];

    // Acceso posicional con variable
    idx := 0$s;
    d[idx] := 42$s;
    r_short := d[idx];
    pout(r_short);

    // Invocacion de metodos heredados
    r_short := d.m_base1();
    r_short := d.m_base2(5$s);
    r_short := d.m_propio();

    // Invocacion con visibilidad entre clases
    cli.m_cliente(b);

    // Invocacion en jerarquia en diamante
    r_short := dia.m_raiz();
    r_short := dia.m_comun();
    r_short := dia.m_rama2();
    r_short := dia.m_propio_dia();

    pout(r_short);
    pout("Fin de pruebas clases ok");
end
