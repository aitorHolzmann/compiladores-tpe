programa_comptime_error

comptime shortint c_entero, c_sin_init;
comptime singlef c_float;
shortint r_entero;
singlef r_float;

begin
    r_entero := 5$s;
    r_float := 2.5;

    // Asignacion de variable de runtime a variable comptime
    c_entero := r_entero;

    // Asignacion de expresion con variables de runtime a comptime
    c_entero := r_entero + 2$s;

    // Uso de comptime antes de ser inicializada
    r_entero := c_sin_init;

    // Incompatibilidad de tipos en asignacion comptime
    c_entero := 3.14;
    c_float := 10$s;

    // Asignacion en expresion de runtime a comptime
    r_entero := c_entero = (r_entero);
end
