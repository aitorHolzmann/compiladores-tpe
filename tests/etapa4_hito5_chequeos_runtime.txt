programa_hito5_chequeos_runtime

shortint caso, a, b, res_int;
singlef f1, f2, f_res, f_conv;

// Funcion para probar Chequeo i: recursion no permitida
shortint FUNCTION contar_rec(shortint n)
begin
  if (n > 0$s)
  begin
    ret(contar_rec(n - 1$s));
  end
  end_if;
  ret(0$s);
end;

begin
  // Cambiar 'caso' (1 a 4) para disparar cada chequeo en tiempo de ejecucion,
  // o dejar en 0$s para ejecutar los casos validos sin error.
  caso := 0$s;

  // Casos base validos
  a := 10$s;
  b := 5$s;
  res_int := a * b;
  pout("Producto valido 10 * 5:");
  pout(res_int);

  f1 := 10.5;
  f2 := 2.5;
  f_res := f1 + f2;
  pout("Suma flotante valida 10.5 + 2.5:");
  pout(f_res);

  f_conv := 42.0;
  res_int := tos(f_conv);
  pout("Conversion tos(42.0) valida:");
  pout(res_int);

  res_int := contar_rec(0$s);
  pout("Llamada sin recursion contar_rec(0):");
  pout(res_int);

  // Chequeo c: Overflow en suma de SINGLEF (> 3.40282347E+38)
  if (caso == 1$s)
  begin
    f1 := 10.0s38 * 2.5;
    f2 := 10.0s38 * 2.0;
    f_res := f1 + f2;
    pout(f_res);
  end
  end_if;

  // Chequeo d: Overflow en producto de SHORTINT (fuera de [-128, 127])
  if (caso == 2$s)
  begin
    a := 50$s;
    b := 3$s;
    res_int := a * b;
    pout(res_int);
  end
  end_if;

  // Chequeo g: Perdida de informacion en tos() (decimales o fuera de rango)
  if (caso == 3$s)
  begin
    f_conv := 10.5;
    res_int := tos(f_conv);
    pout(res_int);
  end
  end_if;

  // Chequeo i: Recursion directa o mutua no permitida
  if (caso == 4$s)
  begin
    res_int := contar_rec(2$s);
    pout(res_int);
  end
  end_if;

  pout("Hito 5: ejecucion finalizada sin errores.");
end
