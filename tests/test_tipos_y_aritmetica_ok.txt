programa_tipos_ok

shortint a, b, c, res_int;
singlef x, y, z, res_flt;

BEGIN
  pout("Constantes:");
  pout(10$s);
  pout(-5$s);
  pout(3.14);
  pout(-0.5);
  pout(1.5s2);

  // Operaciones homogeneas shortint
  a := 10$s;
  b := 3$s;
  res_int := a + b * 2$s - 6$s / b;
  pout(res_int);

  // Asignacion dentro de expresion
  c := b = (4$s) + a;
  pout(b);
  pout(c);

  // Operaciones homogeneas singlef
  x := 12.5;
  y := 2.5;
  z := x / y + 3.0 * y - 1.5;
  pout(z);

  // Conversion explicita tos()
  x := 2.0;
  y := 4.0;
  a := tos(x);
  b := tos(x + y);
  res_flt := x + y;
  c := tos(res_flt) + 5$s;
  pout(a);
  pout(b);
  pout(c);
  pout(tos(res_flt));

  // Asignacion en expresion con singlef
  x := y = (10.0);
  pout(x);
  pout(y);
END
