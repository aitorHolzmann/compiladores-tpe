shortint function suma(shortint p1, shortint p2)
    shortint acumulador;
begin
    acumulador := p1 + p2;
    ret (acumulador);
end;

auto function cuadrado(shortint num)
begin
    ret (num * num);
end;

auto function max_abarcativo(shortint val_entero, singlef val_float)
begin
    if (tos(val_float) > val_entero)
        ret (val_float);
    else
        ret (val_entero);
    end_if;
end;

shortint function sin_retorno(shortint p)
begin
    p := p + 1$s;
end;

// Funcion anidada para verificar ambitos independientes
shortint function f_externa(shortint p_ext)
    shortint function f_interna(shortint p_int)
    begin
        ret (p_int * 2$s);
    end;
    shortint res_int;
begin
    res_int := f_interna(p_ext);
    ret (res_int + 1$s);
end;

begin
    a := 10$s;
    b := 20$s;
    x := 1.5;
    y := 3.0;

    res_short := suma(a, b);
    pout(res_short);

    res_short := cuadrado(5$s);
    pout(res_short);

    res_single := max_abarcativo(a, y);
    pout(res_single);

    res_short := suma(2$s, 3$s) + 4$s;
    pout(res_short);

    res_short := suma(a, b = (5$s));
    pout(res_short);

    res_short := sin_retorno(a);
    pout(res_short);

    res_short := f_externa(10$s);
    pout(res_short);

    pout("Fin de pruebas funciones ok");
end
