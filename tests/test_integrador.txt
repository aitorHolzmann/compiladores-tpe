programa_integrador_grupo20

shortint a, b, c;
singlef f1;
comptime shortint limite_estatico;

// Clases, visibilidad y herencia multiple
class c_base_a
begin
  shortint atr_a export to c_hija;
  shortint metodo_a(shortint p)
  begin
    ret(p + 1$s);
  end export to c_hija;
end;

class c_base_b
begin
  singlef atr_b export to c_hija;
end;

class c_hija import from c_base_a, c_base_b
begin
  extends c_base_a, c_base_b;
  shortint atr_propio;
end;

c_hija obj;

// Funcion ordinaria
shortint FUNCTION duplicar(shortint num)
begin
  ret(num * 2$s);
end;

// Funcion AUTO con inferencia de retorno y conversion tos()
AUTO FUNCTION calcular(shortint x, singlef y)
begin
  ret(tos(y) + x);
end;

begin
  limite_estatico := 5$s;

  a := 2$s;
  f1 := 10.0;

  // Asignacion en expresion y conversion tos()
  b := c = (tos(f1) + a);

  // Acceso posicional a atributos
  obj[0$s] := 20$s;
  obj[a] := 30$s;

  // Invocacion a metodo heredado y funcion auto
  c := obj.metodo_a(a);
  a := calcular(c, f1);

  if (a > limite_estatico)
  begin
    b := duplicar(a);
  end
  else
  begin
    b := 0$s;
  end
  end_if;

  repeat
  begin
    b := b - 1$s;
  end
  until (b <= 0$s);

  pout(b);
  pout(c);

  // Cadena multilinea en sentencia pout
  pout("Integracion Grupo 20
finalizada exitosamente");

end
