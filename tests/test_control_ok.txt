programa_control_ok

shortint a, b, c;
singlef x, y, z;

BEGIN
  a := 10$s;
  b := 5$s;
  c := 0$s;
  x := 1.5;
  y := 3.0;
  z := 0.0;

  // IF simple con shortint
  IF (a >= b)
    c := 1$s;
  END_IF;
  pout(c);

  // IF-ELSE con shortint
  IF (a == b)
    c := 9$s;
  ELSE
    c := 2$s;
  END_IF;
  pout(c);

  // IF-ELSE con singlef
  IF (x < y)
    z := 10.0;
  ELSE
    z := 20.0;
  END_IF;
  pout(z);

  // IF anidados dentro de IF-ELSE
  IF (a != b)
    IF (c == 2$s)
      a := 1$s;
    ELSE
      a := 2$s;
    END_IF;
  ELSE
    a := 3$s;
  END_IF;
  pout(a);

  // REPEAT UNTIL con sentencia simple
  a := 0$s;
  REPEAT
    a := a + 1$s;
  UNTIL (a >= 5$s);
  pout(a);

  // REPEAT UNTIL con bloque BEGIN ... END
  x := 0.0;
  REPEAT
  BEGIN
    x := x + 0.5;
    z := z + 1.0;
  END
  UNTIL (x >= 2.0);
  pout(x);
  pout(z);

  // REPEAT conteniendo IF anidado
  REPEAT
    IF (a < 10$s)
      a := a + 1$s;
    END_IF;
  UNTIL (a >= 10$s);
  pout(a);

  pout("Fin de control de flujo ok");
END
