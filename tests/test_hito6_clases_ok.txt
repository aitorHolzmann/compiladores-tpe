        ret(b1_atr0);

end;

class c_base2

    shortint b2_atr0;

    shortint m_base2(shortint x)

        ret(x + 1$s);
    end;
end;

class c_derivada
begin
    extends c_base1, c_base2;
    shortint der_atr0;

    shortint m_propio()

        ret(100$s);
    end;
end;

begin

    begin
        obj_b1[0$s] := 10$s;
        obj_b1.m_base1();
        ret(0$s);
    end;
end;
